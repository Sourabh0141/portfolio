\newcommand{\resumeheader}[1]{%
  \begin{center}
    {\fontsize{21}{24}\selectfont\bfseries \myname}\par
    \vspace{2.5pt}
    {\small\bfseries #1}\par
    \vspace{2.5pt}
    {\small \mylocation{} \textbar{} \href{tel:\myphonelink}{\myphone} \textbar{}
      \href{mailto:\myemail}{\myemail}}\par
    \vspace{1pt}
    {\small \href{https://\myportfolio}{\myportfolio} \textbar{}
      \href{https://www.\mylinkedin}{\mylinkedin} \textbar{}
      \href{https://\mygithub}{\mygithub}}\par
  \end{center}
  \vspace{-7pt}}
